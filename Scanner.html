// 1. تشغيل الكاميرا بإعدادات فائقة السرعة للقط الفوري const html5QrCode = new Html5Qrcode("reader"); const config = { fps: 25, // فحص فائق السرعة (25 إطار في الثانية) qrbox: (viewfinderWidth, viewfinderHeight) => { // مساحة التقاط واسعة تشمل 80% من الكادر لسرعة الفوكس const minEdge = Math.min(viewfinderWidth, viewfinderHeight); return { width: Math.floor(minEdge * 0.8), height: Math.floor(minEdge * 0.8) }; }, aspectRatio: 1.0 }; html5QrCode.start( { facingMode: "environment" }, config, onScanSuccess, onScanFailure ); let isProcessing = false; async function onScanSuccess(decodedText) { if (isProcessing) return; isProcessing = true; // إيقاف الفحص مؤقتاً أثناء عرض بيانات التذكرة html5QrCode.pause(true); const ticketCode = decodedText.trim().toUpperCase(); await verifyAndShowTicket(ticketCode); } function onScanFailure(error) { // تجاهل إطارات الفحص الفارغة } // 2. التحقق وعرض النافذة الثابتة (لا تختفي إلا بالضغط) async function verifyAndShowTicket(code) { try { // البحث عن المقعد المرتبط بالكود const snapshot = await get(ref(db, 'bookings')); const bookings = snapshot.val() || {}; let foundSeat = null; let guestData = null; for (const [seatId, data] of Object.entries(bookings)) { if (data.code === code) { foundSeat = seatId; guestData = data; break; } } if (!foundSeat) { // كود غير موجود await Swal.fire({ icon: 'error', title: 'تذكرة غير صالحة ❌', text: 'هذا الكود غير مسجل في النظام أو تم إلغاؤه.', confirmButtonText: 'مسح تذكرة أخرى', confirmButtonColor: '#e74c3c', allowOutsideClick: false }); } else if (guestData.status === 'attended') { // كود مستخدم مسبقاً await Swal.fire({ icon: 'warning', title: 'مستخدمة مسبقاً ⚠️', html: `تم تسجيل دخول هذه التذكرة بالفعل!

المقعد: ${foundSeat}
الاسم: ${guestData.name}`, confirmButtonText: 'تخطي ومتابعة', confirmButtonColor: '#f39c12', allowOutsideClick: false }); } else { // تذكرة صحيحة - تسجيل الحضور وعرض النافذة الثابتة await update(ref(db, `bookings/${foundSeat}`), { status: 'attended', attendedAt: new Date().toISOString() }); await Swal.fire({ icon: 'success', title: 'تم تأكيد الدخول ✅', html: `

رقم المقعد: ${foundSeat}

اسم الضيف: ${guestData.name}

كود التذكرة: ${code}

`, confirmButtonText: 'موافق (التذكرة التالية)', confirmButtonColor: '#2c3e50', allowOutsideClick: false }); } } catch (err) { await Swal.fire('خطأ', 'حدثت مشكلة أثناء الاتصال بقاعدة البيانات', 'error'); } finally { // استئناف الكاميرا للمسح التالي بعد الضغط على الزرار html5QrCode.resume(); isProcessing = false; } }
